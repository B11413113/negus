/* Includes ------------------------------------------------------------------*/
#include "main.h"         // 引入主標頭檔，包含 STM32 HAL 庫與核心定義
#include <string.h>       // 引入字串處理函式庫（如 strcmp, strncpy, strtok）
#include <stdlib.h>       // 引入標準函式庫（如 atoi 字串轉整數）
#include <stdio.h>        // 引入標準輸入輸出函式庫（如 sprintf）

/* Private variables ---------------------------------------------------------*/
UART_HandleTypeDef hlpuart1; // 宣告 LPUART1 通訊控制代碼結構體
TIM_HandleTypeDef htim3;     // 宣告 TIM3 計時器控制代碼結構體（用於 PWM 呼吸燈）

static uint8_t rxByte;       // 宣告單一接收位元組變數（保留使用）
static char lineBuffer[32];  // 宣告接收指令的緩衝區字元陣列，最大長度 32
static uint8_t lineLength = 0U; // 記錄目前緩衝區內已經接收到的字元數量

/* 系統運行狀態列舉 */
typedef enum {
    MODE_UNARMED = 0, // 狀態 0：未啟動鎖定狀態（開機或 Reset 後預設，需輸入 ON 解鎖）
    MODE_OFF,        // 狀態 1：已解鎖但目前停止中（燈號與蜂鳴器關閉）
    MODE_LEFT,       // 狀態 2：左方向燈模式
    MODE_RIGHT,      // 狀態 3：右方向燈模式
    MODE_HAZARD      // 狀態 4：雙黃警示燈（雙閃）模式
} SystemMode_t;

volatile SystemMode_t current_mode = MODE_UNARMED; // 宣告全域系統模式變數，初值為 UNARMED（上電或重置後鎖定）

/* 參數變數與預設值（當完全沒設定過時的初始預設值） */
uint32_t blink_interval = 500;    // 閃爍週期初始值（毫秒）
uint32_t target_brightness = 500; // 目標亮度初始值（0~999，對應 50% 亮度）
uint32_t buzzer_interval = 500;   // 蜂鳴器嗶聲間隔初始值（毫秒）

/* 呼吸燈與計時輔助變數 */
uint32_t last_blink_tick = 0;     // 記錄上一次呼吸燈更新的時間點（tick）
uint32_t last_buzzer_tick = 0;    // 記錄上一次蜂鳴器切換的時間點（tick）
uint32_t raw_step = 0;            // 呼吸燈當前的進度步數（0 到 1000）
int8_t step_dir = 1;              // 呼吸燈變化方向（1 代表變亮，-1 代表變暗）
uint8_t delay_cnt = 0;            // 呼吸燈速度分頻計數器
uint8_t buzzer_state = 0;         // 記錄蜂鳴器當前的輸出高低準位狀態（0 或 1）

/* Private function prototypes -----------------------------------------------*/
void SystemClock_Config(void);    // 系統時脈設定函式宣告
static void MX_GPIO_Init(void);   // GPIO 初始化函式宣告
static void MX_LPUART1_UART_Init(void); // LPUART1 初始化函式宣告
static void MX_TIM3_Init(void);   // TIM3 PWM 初始化函式宣告
static void ProcessReceivedByte(uint8_t data); // 字元組合處理函式宣告
static void ExecuteCommand(const char *command); // 指令解析與執行函式宣告

/**
  * @brief  解析並執行從序列埠接收到的控制指令
  * @param  command: 接收到的字串指令
  */
static void ExecuteCommand(const char *command)
{
    char cmd_copy[32];                    // 宣告區域字元陣列用作指令複製副本
    strncpy(cmd_copy, command, sizeof(cmd_copy) - 1); // 將傳入指令複製到副本中
    cmd_copy[sizeof(cmd_copy) - 1] = '\0'; // 確保字串結尾有 Null 字元

    // 徹底清除結尾殘留的換行字元或空白，避免字串比對失敗
    size_t len = strlen(cmd_copy);
    while (len > 0 && (cmd_copy[len - 1] == '\r' || cmd_copy[len - 1] == '\n' || cmd_copy[len - 1] == ' ')) {
        cmd_copy[len - 1] = '\0';
        len--;
    }

    char *token = strtok(cmd_copy, " ");  // 使用空白字元切割取得第一個詞（指令動作）
    if (token == NULL) return;            // 若詞為空則直接返回不處理

    /* 0. 處理 ON 啟動/解鎖指令（任何時候輸入 ON 都能解鎖並進入待命 OFF 狀態） */
    if (strcmp(token, "ON") == 0)
    {
        current_mode = MODE_OFF;          // 解鎖系統，進入待命關閉模式
        __HAL_TIM_SET_COMPARE(&htim3, TIM_CHANNEL_3, 0); // 將左燈歸零
        __HAL_TIM_SET_COMPARE(&htim3, TIM_CHANNEL_4, 0); // 將右燈歸零
        HAL_GPIO_WritePin(GPIOC, GPIO_PIN_13, GPIO_PIN_RESET); // 關閉蜂鳴器

        char *msg = "STATUS: SYSTEM UNLOCKED & READY (OFF MODE)\r\n";
        HAL_UART_Transmit(&hlpuart1, (uint8_t*)msg, strlen(msg), 100);
        return;
    }

    /* 如果系統仍處於未解鎖狀態（MODE_UNARMED），拒絕執行其他所有指令 */
    if (current_mode == MODE_UNARMED)
    {
        char *msg = "ERROR: System is locked. Please enter 'ON' first.\r\n";
        HAL_UART_Transmit(&hlpuart1, (uint8_t*)msg, strlen(msg), 100);
        return;
    }

    /* 1. 處理 OFF 停止/歸零指令 */
    if (strcmp(token, "OFF") == 0)
    {
        current_mode = MODE_OFF;          // 將系統模式切換為待命 OFF
        __HAL_TIM_SET_COMPARE(&htim3, TIM_CHANNEL_3, 0); // 將 TIM3 通道 3（左燈）PWM 歸零
        __HAL_TIM_SET_COMPARE(&htim3, TIM_CHANNEL_4, 0); // 將 TIM3 通道 4（右燈）PWM 歸零
        HAL_GPIO_WritePin(GPIOC, GPIO_PIN_13, GPIO_PIN_RESET); // 將 PC13 蜂鳴器準位拉低關閉

        char *msg = "STATUS: STANDBY (OFF)\r\n"; // 定義回傳訊息字串
        HAL_UART_Transmit(&hlpuart1, (uint8_t*)msg, strlen(msg), 100); // 透過 LPUART1 傳送訊息
    }
    /* 2. 處理 LEFT, RIGHT, HAZARD 模式指令 */
    else
    {
        if (strcmp(token, "LEFT") == 0) {
            if (current_mode != MODE_LEFT) { raw_step = 0; step_dir = 1; } // 切換模式時重置呼吸進度
            current_mode = MODE_LEFT;
        }
        else if (strcmp(token, "RIGHT") == 0) {
            if (current_mode != MODE_RIGHT) { raw_step = 0; step_dir = 1; } // 切換模式時重置呼吸進度
            current_mode = MODE_RIGHT;
        }
        else if (strcmp(token, "HAZARD") == 0) {
            if (current_mode != MODE_HAZARD) { raw_step = 0; step_dir = 1; } // 切換模式時重置呼吸進度
            current_mode = MODE_HAZARD;
        }
        else
        {
            char *msg = "ERROR: Unknown Command\r\n";
            HAL_UART_Transmit(&hlpuart1, (uint8_t*)msg, strlen(msg), 100);
            return;                      // 若皆不符合則返回不處理
        }

        /* ================================================================= */
        /* 核心功能：逐一獨立解析參數 [週期ms] [亮度%] [蜂鳴器ms]            */
        /* 說明：使用獨立的 if 判斷。若使用者未輸入該參數(即為 NULL)，        */
        /*       則「完全不修改」對應變數，直接保留上一次設定的數值！          */
        /* ================================================================= */
        char *p1 = strtok(NULL, " ");     // 切割取得第二個參數（閃爍週期）
        char *p2 = strtok(NULL, " ");     // 切割取得第三個參數（亮度百分比）
        char *p3 = strtok(NULL, " ");     // 切割取得第四個參數（蜂鳴器間隔）

        // 參數一：閃爍週期檢查與更新（允許極限到 1ms）
        if (p1 != NULL) {
            blink_interval = atoi(p1);
            if (blink_interval < 1) blink_interval = 1;
        }

        // 參數二：亮度百分比檢查與更新
        if (p2 != NULL) {
            uint32_t bri_percent = atoi(p2);
            if (bri_percent > 100) bri_percent = 100;
            target_brightness = (bri_percent * 999) / 100;
        }

        // 參數三：蜂鳴器間隔檢查與更新（允許極限到 1ms）
        if (p3 != NULL) {
            buzzer_interval = atoi(p3);
            if (buzzer_interval < 1) buzzer_interval = 1;
        }

        /* 回傳設定成功訊息與目前實際生效的參數狀態 */
        char msg[64];                     // 宣告回傳訊息緩衝陣列
        sprintf(msg, "OK: Mode Set (Int:%d, Bri:%d%%, Buz:%d)\r\n",
                (int)blink_interval, (int)((target_brightness * 100) / 999), (int)buzzer_interval);
        HAL_UART_Transmit(&hlpuart1, (uint8_t*)msg, strlen(msg), 100); // 傳送狀態訊息給終端機
    }
}

/**
  * @brief  字元組合處理器，遇到換行符號時觸發指令執行
  */
static void ProcessReceivedByte(uint8_t data)
{
    if (data == '\r' || data == '\n')     // 若偵測到 Carriage Return 或 Newline 換行字元
    {
        if (lineLength > 0)               // 且緩衝區內有累積字元
        {
            lineBuffer[lineLength] = '\0'; // 在字串結尾補上結束字元
            ExecuteCommand(lineBuffer);   // 呼叫指令解析函式執行命令
            lineLength = 0U;              // 重置行長度計數器
        }
    }
    else if (lineLength < sizeof(lineBuffer) - 1U) // 若尚未換行且未超過緩衝區大小
    {
        lineBuffer[lineLength++] = (char)data; // 將接收到的字元存入緩衝陣列並將長度加 1
    }
}
/* USER CODE END 0 */

int main(void)
{
  /* 硬體與 HAL 庫初始化 */
  HAL_Init();                    // 初始化 HAL 庫與底層系統計時器
  SystemClock_Config();          // 設定系統主時脈

  MX_GPIO_Init();                // 初始化所有 GPIO 腳位
  MX_LPUART1_UART_Init();        // 初始化 LPUART1 序列埠
  MX_TIM3_Init();                // 初始化 TIM3 計時器

  /* USER CODE BEGIN 2 */
  /* 啟動 TIM3 PWM 通道 3 (PC8, 左燈) 與通道 4 (PC9, 右燈) */
  HAL_TIM_PWM_Start(&htim3, TIM_CHANNEL_3); // 開啟 TIM3 CH3 PWM 輸出
  HAL_TIM_PWM_Start(&htim3, TIM_CHANNEL_4); // 開啟 TIM3 CH4 PWM 輸出

  /* 發送開機鎖定提示訊息給終端機 */
  char *welcomeMsg = "SYSTEM LOCKED: Please enter 'ON' to start\r\n";
  HAL_UART_Transmit(&hlpuart1, (uint8_t*)welcomeMsg, strlen(welcomeMsg), 100); // 透過序列埠送出
  /* USER CODE END 2 */

  while (1)                      // 進入無限迴圈主程式
  {
    /* 非阻塞式檢查 LPUART1 是否收到新資料 (避免卡死主迴圈) */
    if (__HAL_UART_GET_FLAG(&hlpuart1, UART_FLAG_RXNE) != RESET) // 檢查接收暫存器是否不為空
    {
      uint8_t rxData = (uint8_t)(hlpuart1.Instance->RDR & 0xFF); // 讀取接收到的單一資料位元組
      ProcessReceivedByte(rxData); // 送入字元組合處理器
    }

    uint32_t current_tick = HAL_GetTick(); // 取得目前的系統時間 Tick（毫秒）

    /* 若系統處於未解鎖狀態 (MODE_UNARMED)，強制清空所有輸出並跳過燈光邏輯 */
    if (current_mode == MODE_UNARMED)
    {
      __HAL_TIM_SET_COMPARE(&htim3, TIM_CHANNEL_3, 0);
      __HAL_TIM_SET_COMPARE(&htim3, TIM_CHANNEL_4, 0);
      HAL_GPIO_WritePin(GPIOC, GPIO_PIN_13, GPIO_PIN_RESET);
      continue; // 直接進入下一個迴圈
    }

    /* 當系統處於動作模式（非 UNARMED 且非 OFF）時執行燈光與蜂鳴器邏輯 */
    if (current_mode != MODE_OFF)
    {
      /* 1. 呼吸燈平滑漸層控制 */
      if (current_tick - last_blink_tick >= 1) // 每隔 1 毫秒執行一次亮度計算
      {
        last_blink_tick = current_tick;      // 更新上次執行時間

        /* 根據 blink_interval 動態調整亮度變化的步進速度 */
        uint32_t speed_div = blink_interval / 10;
        if (speed_div < 1) speed_div = 1;

        delay_cnt++;                          // 累加計數器
        if (delay_cnt >= speed_div)           // 當達到分頻門檻時才改變亮度步階
        {
            delay_cnt = 0;                    // 重置分頻計數器
            raw_step += step_dir;             // 根據方向增加或減少步數
            if (raw_step >= 1000) { raw_step = 1000; step_dir = -1; } // 若達到最高點(1000)，改為遞減變暗
            else if (raw_step <= 0) { raw_step = 0; step_dir = 1; }   // 若降到最低點(0)，改為遞增變亮

            /* 非線性平方曲線運算，讓呼吸燈亮度變化更符合人眼視覺特性 */
            uint32_t current_breath = (raw_step * raw_step * target_brightness) / 1000000;

            /* 依據當前模式分配 PWM 輸出到左右燈 */
            if (current_mode == MODE_LEFT || current_mode == MODE_HAZARD)
                __HAL_TIM_SET_COMPARE(&htim3, TIM_CHANNEL_3, current_breath); // 設定左燈 PWM 脈衝
            else
                __HAL_TIM_SET_COMPARE(&htim3, TIM_CHANNEL_3, 0);               // 關閉左燈

            if (current_mode == MODE_RIGHT || current_mode == MODE_HAZARD)
                __HAL_TIM_SET_COMPARE(&htim3, TIM_CHANNEL_4, current_breath); // 設定右燈 PWM 脈衝
            else
                __HAL_TIM_SET_COMPARE(&htim3, TIM_CHANNEL_4, 0);               // 關閉右燈
        }
      }

      /* 2. 外接蜂鳴器間隔控制 (PC13) */
      if (current_tick - last_buzzer_tick >= buzzer_interval) // 當達到設定的蜂鳴器間隔時間
      {
        last_buzzer_tick = current_tick;      // 更新上次蜂鳴器切換時間
        buzzer_state = !buzzer_state;         // 反轉蜂鳴器狀態（0 變 1，1 變 0）
        HAL_GPIO_WritePin(GPIOC, GPIO_PIN_13, buzzer_state ? GPIO_PIN_SET : GPIO_PIN_RESET); // 輸出高低準位至 PC13
      }
    }
    else
    {
      /* 狀態為 OFF（已解鎖但待命）時強制清空所有輸出 */
      __HAL_TIM_SET_COMPARE(&htim3, TIM_CHANNEL_3, 0); // 關閉左燈 PWM
      __HAL_TIM_SET_COMPARE(&htim3, TIM_CHANNEL_4, 0); // 關閉右燈 PWM
      HAL_GPIO_WritePin(GPIOC, GPIO_PIN_13, GPIO_PIN_RESET); // 關閉蜂鳴器
    }
  }
}

/**
  * @brief System Clock Configuration (系統時脈配置函式)
  */
void SystemClock_Config(void)
{
  RCC_OscInitTypeDef RCC_OscInitStruct = {0}; // 宣告振盪器初始化結構體
  RCC_ClkInitTypeDef RCC_ClkInitStruct = {0}; // 宣告時脈供應初始化結構體

  HAL_PWREx_ControlVoltageScaling(PWR_REGULATOR_VOLTAGE_SCALE1_BOOST); // 設定核心電壓級別

  RCC_OscInitStruct.OscillatorType = RCC_OSCILLATORTYPE_HSI; // 設定使用內部高速振盪器 (HSI)
  RCC_OscInitStruct.HSIState = RCC_HSI_ON;                   // 開啟 HSI
  RCC_OscInitStruct.HSICalibrationValue = RCC_HSICALIBRATION_DEFAULT; // HSI 預設校準值
  RCC_OscInitStruct.PLL.PLLState = RCC_PLL_ON;               // 開啟 PLL 鎖相迴路
  RCC_OscInitStruct.PLL.PLLSource = RCC_PLLSOURCE_HSI;       // 設定 PLL 來源為 HSI
  RCC_OscInitStruct.PLL.PLLM = RCC_PLLM_DIV4;                // PLL M 除頻係數
  RCC_OscInitStruct.PLL.PLLN = 85;                           // PLL N 倍頻係數
  RCC_OscInitStruct.PLL.PLLP = RCC_PLLP_DIV2;                // PLL P 除頻
  RCC_OscInitStruct.PLL.PLLQ = RCC_PLLQ_DIV2;                // PLL Q 除頻
  RCC_OscInitStruct.PLL.PLLR = RCC_PLLR_DIV2;                // PLL R 除頻
  if (HAL_RCC_OscConfig(&RCC_OscInitStruct) != HAL_OK)       // 套用振盪器設定並檢查是否錯誤
  {
    Error_Handler();                                         // 若失敗則進入錯誤處理
  }

  RCC_ClkInitStruct.ClockType = RCC_CLOCKTYPE_HCLK|RCC_CLOCKTYPE_SYSCLK
                              |RCC_CLOCKTYPE_PCLK1|RCC_CLOCKTYPE_PCLK2; // 設定要配置的時脈類型
  RCC_ClkInitStruct.SYSCLKSource = RCC_SYSCLKSOURCE_PLLCLK;     // 設定系統時脈源為 PLL
  RCC_ClkInitStruct.AHBCLKDivider = RCC_SYSCLK_DIV1;         // AHB 匯流排不除頻
  RCC_ClkInitStruct.APB1CLKDivider = RCC_HCLK_DIV1;          // APB1 匯流排不除頻
  RCC_ClkInitStruct.APB2CLKDivider = RCC_HCLK_DIV1;          // APB2 匯流排不除頻

  if (HAL_RCC_ClockConfig(&RCC_ClkInitStruct, FLASH_LATENCY_4) != HAL_OK) // 套用時脈並設定快閃記憶體等待週期
  {
    Error_Handler();                                         // 若失敗則進入錯誤處理
  }
}

/**
  * @brief LPUART1 Initialization Function (低功耗序列埠 1 初始化)
  */
static void MX_LPUART1_UART_Init(void)
{
  hlpuart1.Instance = LPUART1;                  // 指定實例為 LPUART1
  hlpuart1.Init.BaudRate = 115200;              // 設定鮑率為 115200
  hlpuart1.Init.WordLength = UART_WORDLENGTH_8B;// 資料位元長度 8 bits
  hlpuart1.Init.StopBits = UART_STOPBITS_1;     // 停止位元 1 bit
  hlpuart1.Init.Parity = UART_PARITY_NONE;      // 不啟用同位檢查
  hlpuart1.Init.Mode = UART_MODE_TX_RX;         // 啟用傳送與接收模式
  hlpuart1.Init.HwFlowCtl = UART_HWCONTROL_NONE;// 不啟用硬體流控
  hlpuart1.Init.OneBitSampling = UART_ONE_BIT_SAMPLE_DISABLE; // 停用單位元取樣
  hlpuart1.Init.ClockPrescaler = UART_PRESCALER_DIV1; // 時脈預分頻為 1
  hlpuart1.AdvancedInit.AdvFeatureInit = UART_ADVFEATURE_NO_INIT; // 進階功能不初始化
  if (HAL_UART_Init(&hlpuart1) != HAL_OK)       // 初始化 LPUART1 並檢查是否成功
  {
    Error_Handler();                            // 若失敗則進入錯誤處理
  }
}

/**
  * @brief TIM3 Initialization Function (計時器 3 PWM 輸出初始化)
  */
static void MX_TIM3_Init(void)
{
  TIM_MasterConfigTypeDef sMasterConfig = {0};  // 宣告計時器主控設定結構體
  TIM_OC_InitTypeDef sConfigOC = {0};           // 宣告輸出比較（PWM）設定結構體

  htim3.Instance = TIM3;                        // 指定實例為 TIM3
  htim3.Init.Prescaler = 169;                   // 設定預分頻器
  htim3.Init.CounterMode = TIM_COUNTERMODE_UP;  // 設定向上計數模式
  htim3.Init.Period = 999;                      // 設定自動重載周期（ARR = 999）
  htim3.Init.ClockDivision = TIM_CLOCKDIVISION_DIV1; // 時脈分割不分頻
  htim3.Init.AutoReloadPreload = TIM_AUTORELOAD_PRELOAD_DISABLE; // 停用自動重載預載入
  if (HAL_TIM_PWM_Init(&htim3) != HAL_OK)       // 初始化 PWM 並檢查是否成功
  {
    Error_Handler();                            // 若失敗則進入錯誤處理
  }
  sMasterConfig.MasterOutputTrigger = TIM_TRGO_RESET; // 設定主輸出觸發為重置
  sMasterConfig.MasterSlaveMode = TIM_MASTERSLAVEMODE_DISABLE; // 停用主從模式
  if (HAL_TIMEx_MasterConfigSynchronization(&htim3, &sMasterConfig) != HAL_OK) // 套用主設定
  {
    Error_Handler();                            // 若失敗則進入錯誤處理
  }
  sConfigOC.OCMode = TIM_OCMODE_PWM1;           // 設定 PWM 模式 1
  sConfigOC.Pulse = 0;                          // 初始脈衝寬度（佔空比）設為 0
  sConfigOC.OCPolarity = TIM_OCPOLARITY_HIGH;   // 輸出極性為高電位有效
  sConfigOC.OCFastMode = TIM_OCFAST_DISABLE;    // 停用快速模式
  if (HAL_TIM_PWM_ConfigChannel(&htim3, &sConfigOC, TIM_CHANNEL_3) != HAL_OK) // 設定通道 3（左燈）
  {
    Error_Handler();                            // 若失敗則進入錯誤處理
  }
  if (HAL_TIM_PWM_ConfigChannel(&htim3, &sConfigOC, TIM_CHANNEL_4) != HAL_OK) // 設定通道 4（右燈）
  {
    Error_Handler();                            // 若失敗則進入錯誤處理
  }
  HAL_TIM_MspPostInit(&htim3);                  // 呼叫底層腳位與外設腳位對應初始化
}

/**
  * @brief GPIO Initialization Function (通用輸入輸出腳位初始化)
  */
static void MX_GPIO_Init(void)
{
  GPIO_InitTypeDef GPIO_InitStruct = {0};       // 宣告 GPIO 初始化設定結構體

  __HAL_RCC_GPIOC_CLK_ENABLE();                 // 啟用 GPIOC 連接埠時脈
  __HAL_RCC_GPIOF_CLK_ENABLE();                 // 啟用 GPIOF 連接埠時脈
  __HAL_RCC_GPIOA_CLK_ENABLE();                 // 啟用 GPIOA 連接埠時脈
  __HAL_RCC_GPIOB_CLK_ENABLE();                 // 啟用 GPIOB 連接埠時脈

  HAL_GPIO_WritePin(GPIOC, GPIO_PIN_13, GPIO_PIN_RESET); // 預先將 PC13 輸出設為低電位

  GPIO_InitStruct.Pin = GPIO_PIN_13;            // 指定設定的腳位為 Pin 13（蜂鳴器）
  GPIO_InitStruct.Mode = GPIO_MODE_OUTPUT_PP;   // 設定為推挽輸出模式
  GPIO_InitStruct.Pull = GPIO_NOPULL;           // 不啟用上下拉電阻
  GPIO_InitStruct.Speed = GPIO_SPEED_FREQ_LOW;  // 設定輸出速度為低頻
  HAL_GPIO_Init(GPIOC, &GPIO_InitStruct);       // 初始化 GPIOC 13 腳位
}

/**
  * @brief  Error Handler (系統錯誤處理函式)
  */
void Error_Handler(void)
{
  __disable_irq();                              // 關閉所有中斷
  while (1) {}                                  // 進入無窮迴圈停滯，便於除錯
}
